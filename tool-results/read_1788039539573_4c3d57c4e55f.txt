   392→  // Grades & GPA Calculation
   393→  val allGrades: StateFlow<List<StudentGrade>> = repository.allGrades
   394→    .stateIn(viewModelScope, SharingStarted.WhileSubscribed(5000), emptyList())
   395→
   396→  val calculatedGPA: StateFlow<Double> = allGrades.map { gradesList ->
   397→    if (gradesList.isEmpty()) 0.0
   398→    else {
   399→      var totalCoeff = 0.0
   400→      var weightedSum = 0.0
   401→      gradesList.forEach { g ->
   402→        val modAvg = (g.continuousScore * 0.4) + (g.examScore * 0.6)
   403→        weightedSum += (modAvg * g.coefficient)
   404→        totalCoeff += g.coefficient
   405→      }
   406→      if (totalCoeff > 0) Math.round((weightedSum / totalCoeff) * 100.0) / 100.0 else 0.0
   407→    }
   408→  }.stateIn(viewModelScope, SharingStarted.WhileSubscribed(5000), 0.0)

   916→  private val _isAuthenticated = MutableStateFlow(true)
   917→  val isAuthenticated: StateFlow<Boolean> = _isAuthenticated.asStateFlow()
   918→
   919→  private val _isAuthLoading = MutableStateFlow(false)
   920→  val isAuthLoading: StateFlow<Boolean> = _isAuthLoading.asStateFlow()
   921→
   922→  private val _authStatusMessage = MutableStateFlow<String?>(null)
   923→  val authStatusMessage: StateFlow<String?> = _authStatusMessage.asStateFlow()
   924→
   925→  fun loginUser(email: String, password: String, onSuccess: () -> Unit) {
   926→    viewModelScope.launch {
   927→      _isAuthLoading.value = true
   928→      _authStatusMessage.value = null
   929→      
   930→      // Try Supabase auth in background
   931→      val supabaseResult = supabaseSyncService.signInWithEmail(email, password)
   932→      
   933→      // Look up user or match roles
   934→      val users = repository.allUsers.first()
   935→      val matchedUser = users.find { it.email.equals(email.trim(), ignoreCase = true) }
   936→      
   937→      val roleDetermined = when {
   938→        email.contains("admin") || email.contains("owner") -> "OWNER"
   939→        email.contains("specialty") -> "SPECIALTY_ADMIN"
   940→        email.contains("delegate") || email.contains("rep") -> "REPRESENTATIVE"
   941→        matchedUser != null -> matchedUser.role
   942→        else -> "STUDENT"
   943→      }
   944→      
   945→      val currentProfile = repository.studentProfile.first()
   946→      val updatedProfile = (currentProfile ?: StudentProfile()).copy(
   947→        email = email.trim(),
   948→        fullName = matchedUser?.fullName ?: (if (roleDetermined == "OWNER") "المالك (Super Admin)" else if (roleDetermined == "SPECIALTY_ADMIN") "مسؤول التخصص" else if (roleDetermined == "REPRESENTATIVE") "ممثل الفوج 03" else "طالب جامعي"),
   949→        userRole = roleDetermined,
   950→        groupNumber = matchedUser?.groupNumber ?: (currentProfile?.groupNumber ?: "الفوج 03"),
   951→        isConfigured = true
   952→      )
   953→      
   954→      repository.updateStudentProfile(updatedProfile)
   955→      _isAuthenticated.value = true
   956→      _isAuthLoading.value = false
   957→      _currentScreen.value = ScreenRoute.HOME
   958→      onSuccess()
   959→    }
   960→  }
