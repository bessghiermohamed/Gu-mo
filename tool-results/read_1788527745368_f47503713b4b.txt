   256→function AddSlotDialog({

   411→function ImageSchedule() {
   412→  const { t } = useI18n();
   413→  const [imageUrl, setImageUrl] = React.useState<string | null>(null);
   414→
   415→  function handleUpload(e: React.ChangeEvent<HTMLInputElement>) {
   416→    const file = e.target.files?.[0];
   417→    if (!file) return;
   418→    if (!["image/png", "image/jpeg", "application/pdf"].includes(file.type)) {
   419→      toast.error("صيغة غير مدعومة. استخدم PNG أو JPG أو PDF.");
   420→      return;
   421→    }
   422→    if (file.size > 10 * 1024 * 1024) {
   423→      toast.error("حجم الملف يتجاوز 10 ميغابايت.");
   424→      return;
   425→    }
   426→    const reader = new FileReader();
   427→    reader.onload = () => {
   428→      setImageUrl(reader.result as string);
   429→      toast.success("تم تحميل الصورة — تُعرض على هذا الجهاز أثناء الجلسة الحالية.");
   430→    };
   431→    reader.readAsDataURL(file);
   432→  }
   433→
   434→  return (
   435→    <Card className="p-6">
   436→      <div className="space-y-4">
   437→        <div>
   438→          <h3 className="font-bold text-sm mb-1">{t("schedule.uploadImage")}</h3>
   439→          <p className="text-xs text-muted-foreground">{t("schedule.imageHint")}</p>
   440→        </div>
   441→        {imageUrl ? (
   442→          <div className="space-y-3">
   443→            <img
   444→              src={imageUrl}
   445→              alt="Schedule"
   446→              className="w-full rounded-xl border border-border"
   447→            />
   448→            <Button variant="outline" size="sm" onClick={() => setImageUrl(null)}>
   449→              تغيير الصورة
   450→            </Button>
   451→          </div>
   452→        ) : (
   453→          <label className="block">
   454→            <div className="border-2 border-dashed border-border rounded-xl p-8 text-center cursor-pointer hover:border-primary/50 transition-colors">
   455→              <Upload className="w-8 h-8 mx-auto text-muted-foreground mb-2" />
   456→              <p className="text-sm font-medium">اضغط لاختيار صورة</p>
   457→              <p className="text-xs text-muted-foreground mt-1">
   458→                PNG, JPG, PDF — حتى 10MB
   459→              </p>
   460→            </div>
   461→            <input
   462→              type="file"
   463→              accept="image/png,image/jpeg,application/pdf"
   464→              onChange={handleUpload}
   465→              className="hidden"
   466→            />
   467→          </label>
   468→        )}
   469→      </div>
   470→    </Card>
   471→  );
   472→}

   591→// =====================================================
   592→// round 9 — Personal attendance & absence tracking (spec §14–§19)
   593→//
   594→// STRICTLY PERSONAL: the API (/api/attendance) only ever touches the
   595→// logged-in user's OWN records — supervisors have no administrative
   596→// access, no approval workflow, and no way to see or edit a student's
   597→// absences. This is unofficial personal bookkeeping only.

   599→// Layout per spec §16: course name + absence count + "+" button,
   600→// simple and compact. Tapping a row expands its records (deletable).
   601→// =====================================================
   602→interface AttendanceCourse {
   603→  name: string;
   604→  count: number;
   605→  records: Array<{ id: number; moduleName: string; date: string; createdAt: string }>;
   606→}
   607→
   608→function MyAttendance() {
   609→  const [courses, setCourses] = React.useState<AttendanceCourse[]>([]);
   610→  const [total, setTotal] = React.useState(0);
   611→  const [loading, setLoading] = React.useState(true);
   612→  const [expanded, setExpanded] = React.useState<string | null>(null);
   613→  const [addFor, setAddFor] = React.useState<string | null>(null);
   614→  const [deletingId, setDeletingId] = React.useState<number | null>(null);
   615→
   616→  const fetchAttendance = React.useCallback(async () => {
   617→    setLoading(true);
   618→    try {
   619→      const res = await fetch("/api/attendance", { cache: "no-store" });
   620→      const data = await res.json();
   621→      setCourses(data.courses ?? []);
   622→      setTotal(data.totalAbsences ?? 0);
   623→    } catch {
   624→      // silent
   625→    } finally {
   626→      setLoading(false);
   627→    }
   628→  }, []);
   629→
   630→  React.useEffect(() => { fetchAttendance(); }, [fetchAttendance]);
   631→
   632→  async function handleDelete(id: number) {
   633→    setDeletingId(id);
   634→    try {
   635→      const res = await fetch(`/api/attendance?id=${id}`, { method: "DELETE" });
   636→      const data = await res.json().catch(() => ({}));
   637→      if (!res.ok) { toast.error(data.error ?? "فشل الحذف"); return; }
   638→      toast.success("تم حذف سجل الغياب");
   639→      fetchAttendance();
   640→    } catch {
   641→      toast.error("فشل الاتصال");
   642→    } finally {
   643→      setDeletingId(null);
   644→    }
   645→  }
   646→
   647→  const absenceLabel = (n: number) =>
   648→    n === 0 ? "لا غياب" : n === 1 ? "غياب واحد" : n === 2 ? "غيابان" : `${n} غيابات`;
   649→
   650→  return (
   651→    <div className="space-y-4">
   652→      <Card className="p-4 bg-primary/5 border-primary/20">
   653→        <div className="flex items-center gap-3">
   654→          <div className="w-10 h-10 rounded-xl bg-primary/15 text-primary flex items-center justify-center shrink-0">
   655→            <CalendarX className="w-5 h-5" />
   656→          </div>
   657→          <div className="flex-1">
   658→            <h2 className="font-black text-sm">تتبع الغيابات — شخصي وغير رسمي</h2>
   659→            <p className="text-xs text-muted-foreground mt-0.5">
   660→              سجلك الخاص فقط: {absenceLabel(total)} • لا يطّلع عليه أي مشرف — البيانات تُعدَّل منك وحدك
   661→            </p>
   662→          </div>
   663→          <Button size="sm" onClick={() => setAddFor(null)}>
   664→            <Plus className="w-4 h-4 ml-1" />غياب
   665→          </Button>
   666→        </div>
   667→      </Card>
   668→
   669→      {loading ? (
   670→        <Card className="p-8 text-center">
   671→          <Loader2 className="w-6 h-6 mx-auto animate-spin text-muted-foreground mb-2" />
   672→          <p className="text-sm text-muted-foreground">جاري التحميل...</p>
   673→        </Card>
   674→      ) : courses.length === 0 ? (
   675→        <Card className="p-8 text-center bg-muted/30 border-dashed">
   676→          <CalendarX className="w-10 h-10 mx-auto text-muted-foreground mb-3" />
   677→          <h3 className="font-bold text-sm mb-1">لا توجد مقاييس بعد</h3>
   678→          <p className="text-xs text-muted-foreground">
   679→            ستظهر مقاييسك هنا تلقائياً بمجرد إضافتها من لوحة الإدارة — يمكنك تسجيل أول غياب بزر «+».
   680→          </p>
   681→        </Card>
   682→      ) : (
   683→        <Card className="p-4 space-y-1.5">
   684→          {courses.map((course) => {
   685→            const isOpen = expanded === course.name;
   686→            return (
   687→              <div key={course.name} className="rounded-lg border border-border/60 overflow-hidden">
   688→                <div
   689→                  className="flex items-center gap-2 p-3 cursor-pointer hover:bg-muted/40 transition-colors"
   690→                  onClick={() => setExpanded(isOpen ? null : course.name)}
   691→                  role="button"
   692→                  aria-expanded={isOpen}
   693→                >
   694→                  {course.count > 0 ? (
   695→                    <button
   696→                      onClick={(e) => { e.stopPropagation(); setAddFor(course.name); }}
   697→                      className="w-8 h-8 rounded-lg bg-primary/10 text-primary hover:bg-primary/20 flex items-center justify-center shrink-0 transition-colors"
   698→                      aria-label={`إضافة غياب لـ ${course.name}`}
   699→                      title="إضافة غياب"
   700→                    >
   701→                      <Plus className="w-4 h-4" />
   702→                    </button>
   703→                  ) : (
   704→                    <button
   705→                      onClick={(e) => { e.stopPropagation(); setAddFor(course.name); }}
   706→                      className="w-8 h-8 rounded-lg bg-muted text-muted-foreground hover:bg-primary/10 hover:text-primary flex items-center justify-center shrink-0 transition-colors"
   707→                      aria-label={`إضافة غياب لـ ${course.name}`}
   708→                      title="إضافة غياب"
   709→                    >
   710→                      <Plus className="w-4 h-4" />
   711→                    </button>

   713→                  <span className="font-bold text-sm flex-1 min-w-0 truncate">{course.name}</span>

   753→      {addFor !== null && (

   765→/**
   766→ * Add-absence modal (spec §17): student selects course + date → record
   767→ * is created and the count for that course updates immediately.
   768→ */
   769→function AddAbsenceDialog({
   770→  courses, initialCourse, onClose, onSaved,
   771→}: {
   772→  courses: string[];
   773→  initialCourse?: string;
   774→  onClose: () => void;
   775→  onSaved: () => void;
   776→}) {
   777→  const [course, setCourse] = React.useState(initialCourse ?? (courses[0] ?? ""));
   778→  const [customCourse, setCustomCourse] = React.useState("");
   779→  const [date, setDate] = React.useState(new Date().toISOString().split("T")[0]);
   780→  const [saving, setSaving] = React.useState(false);
   781→  const effectiveCourse = course === "__custom__" ? customCourse : course;
   782→
   783→  async function handleSave() {
   784→    if (!effectiveCourse.trim()) { toast.error("اختر/اكتب اسم المقياس"); return; }
   785→    if (!/^\d{4}-\d{2}-\d{2}$/.test(date)) { toast.error("اختر التاريخ"); return; }
   786→    setSaving(true);
   787→    try {
   788→      const res = await fetch("/api/attendance", {
   789→        method: "POST",
   790→        headers: { "Content-Type": "application/json" },
   791→        body: JSON.stringify({ moduleName: effectiveCourse.trim(), date }),
   792→      });
   793→      const data = await res.json();
   794→      if (!res.ok) { toast.error(data.error ?? "فشل الحفظ"); return; }
   795→      toast.success("تم تسجيل الغياب");
   796→      onSaved();
   797→    } catch {
   798→      toast.error("فشل الاتصال");
   799→    } finally {
   800→      setSaving(false);
   801→    }
   802→  }
   803→
   804→  return (
   805→    <Dialog open onOpenChange={(v) => { if (!v) onClose(); }}>
   806→      <DialogContent>
   807→        <DialogHeader>
   808→          <DialogTitle className="flex items-center gap-2"><CalendarX className="w-5 h-5 text-primary" />تسجيل غياب</DialogTitle>
   809→        </DialogHeader>
   810→        <div className="space-y-3 py-2">
   811→          <div className="space-y-1.5">
   812→            <Label>المقياس</Label>
   813→            {courses.length === 0 ? (
   814→              <Input
   815→                value={course}
   816→                onChange={(e) => setCourse(e.target.value)}
   817→                placeholder="اسم المقياس"
   818→              />
   819→            ) : (
   820→              <select value={course} onChange={(e) => setCourse(e.target.value)} className="flex h-9 w-full rounded-md border border-input bg-transparent px-3 py-1 text-sm">
   821→                {courses.map((c) => <option key={c} value={c}>{c}</option>)}
   822→                <option value="__custom__">— مقياس آخر —</option>
   823→              </select>
   824→            )}
   825→          </div>
   826→          {course === "__custom__" && (
   827→            <div className="space-y-1.5">
   828→              <Label>اسم المقياس</Label>
   829→              <Input
   830→                value={customCourse}
   831→                onChange={(e) => setCustomCourse(e.target.value)}
   832→                placeholder="اكتب اسم المقياس"
   833→              />
   834→            </div>
   835→          )}
   836→          <div className="space-y-1.5">
   837→            <Label>التاريخ</Label>
   838→            <Input
   839→              type="date"
   840→              value={date}
   841→              onChange={(e) => setDate(e.target.value)}
   842→              dir="ltr"
   843→              className="text-right"
   844→            />
   845→          </div>
   846→          <div className="rounded-lg bg-amber-500/10 border border-amber-500/30 p-3 text-xs text-amber-700 dark:text-amber-300 flex items-start gap-2">
   847→            <CheckCircle2 className="w-3.5 h-3.5 mt-0.5 shrink-0" />
   848→            <p>هذا سجل شخصي غير رسمي — لن يراه أي مشرف، ويمكنك حذفه في أي وقت.</p>
   849→          </div>
   850→        </div>
   851→        <DialogFooter>
   852→          <Button variant="outline" onClick={onClose}>إلغاء</Button>
   853→          <Button onClick={handleSave} disabled={saving || !effectiveCourse.trim()}>
   854→            {saving && <Loader2 className="w-4 h-4 ml-1 animate-spin" />}
   855→            تسجيل الغياب
   856→          </Button>
   857→        </DialogFooter>
   858→      </DialogContent>
   859→    </Dialog>
   860→  );
   861→}
