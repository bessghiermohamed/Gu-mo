    57→// =====================================================
    58→// fix (R12-03, P0): the app used to forget the current screen on refresh —
    59→// every reload dumped the student back on HOME, mid-task work was lost and
    60→// no screen could ever be shared/bookmarked. Each route now has a stable
    61→// hash (#/grades, #/courses, …) that is kept in sync so:
    62→//   • refresh / app re-open restores the exact screen
    63→//   • the Android/browser back gesture walks the real visit history
    64→//   • duplicate tab taps no longer poison the back stack (navigate guards)
    65→// =====================================================
    66→const ROUTE_TO_HASH: Record<ScreenRoute, string> = {
    67→  HOME: "home",
    68→  COURSES: "courses",
    69→  SCHEDULE: "schedule",
    70→  EXAMS: "exams",
    71→  GRADES: "grades",
    72→  ASSIGNMENTS: "assignments",
    73→  FILES: "files",
    74→  GROUP: "group",
    75→  BROWSE_GROUPS: "browse-groups",
    76→  PROFILE: "profile",
    77→  SETTINGS: "settings",
    78→  ANNOUNCEMENTS: "announcements",
    79→  ADMIN: "admin",
    80→  TELEGRAM: "telegram",
    81→  COURSE_DETAIL: "course",
    82→  ONBOARDING: "onboarding", // never restored from the URL — gated below
    83→};
    84→
    85→function routeFromHash(hash: string): ScreenRoute {
    86→  const key = hash.replace(/^#\/?/, "").toLowerCase();
    87→  if (!key) return "HOME";
    88→  const found = (Object.entries(ROUTE_TO_HASH) as Array<[ScreenRoute, string]>).find(
    89→    ([, h]) => h === key
    90→  );
    91→  if (!found) return "HOME";
    92→  // never restore INTO onboarding from the URL — the onboarding gate owns it
    93→  if (found[0] === "ONBOARDING") return "HOME";
    94→  return found[0];
    95→}

   223→

   225→    refreshUnreadCount();

   228→      refreshUnreadCount();

   233→

   235→  // and assignment D-2/D-1/D-0 reminders for THIS user, then re-fetch the
   236→  // feed if anything new was created. Triggered on mount + every 10 min —
   237→  // deliberately NOT on the 30s poll loop, so the round-13 polling-cost

   239→  const generateReminders = React.useCallback(async () => {
   240→    if (!user) return;

   250→            .filter(([, v]) => !!v)

   253→        }

   255→        // corrupted storage — generate without completion info
   256→      }
   257→      const res = await fetch("/api/notifications/reminders", {
   258→        method: "POST",
   259→        headers: { "Content-Type": "application/json" },
   260→        body: JSON.stringify({ completedAssignmentIds }),
   261→      });
   262→      if (res.ok) {
   263→        const data = await res.json();
   264→        if (Number(data.created) > 0) refreshNotifications();
   265→      }
   266→    } catch {
   267→      // silent — reminders retry on the next 10-min tick
   268→    }
   269→  }, [user, refreshNotifications]);

   323→      if (prev === "COURSE_DETAIL") return prev;

   329→      window.history.pushState({ talib: "COURSE_DETAIL" }, "", target);

   341→        window.history.pushState({ talib: prev }, "", target);

   426→      } else if (item.type === "content_exam" || item.type === "exam_reminder") {

   437→  const showLoading = React.useCallback((message?: string) => {

   439→    setLoadingMessage(message ?? "");
   440→  }, []);
   441→

   444→    setLoadingMessage("");
   445→  }, []);
   446→
   447→  const shellValue: ShellContextValue = {
   448→    currentScreen,
   449→    navigate,
   450→    navigateBack,

   476→  }
   477→
   478→  // Not authenticated → show login

   487→
   488→  // Needs onboarding — pure render gate (no setState during render):
   489→  // onboarding wins over ANY screen, including a restored deep link.

   491→    return (

   522→  // fix (R12-06, P0): the ADMIN panel used to render for ANY role reached

   538→          <header className="sticky top-0 z-30 bg-background/95 backdrop-blur supports-[backdrop-filter]:bg-background/60 border-b border-border">
   539→            <div className="mx-auto max-w-5xl px-4 py-3 flex items-center gap-3">
   540→              {/* Back button */}
   541→              {effectiveScreen !== "HOME" && (
   542→                <Button
   543→                  variant="ghost"
   544→                  size="icon"
   545→                  onClick={navigateBack}
   546→                  className="shrink-0"
   547→                  aria-label={t("common.back")}
   548→                >
   549→                  <ChevronLeft className={cn("w-5 h-5", dir === "rtl" && "rotate-180")} />
   550→                </Button>
   551→              )}
   552→
   553→              {/* App logo + name */}
   554→              <div className="flex items-center gap-2 flex-1 min-w-0">
   555→                <img
   556→                  src="/talib/icon.svg"
   557→                  alt="Talib"
   558→                  className="w-8 h-8 shrink-0"
   559→                />
   560→                <span className="font-bold text-base truncate">
   561→                  {t("common.appName")}
   562→                </span>
   563→              </div>
   564→
   565→              {/* Action buttons */}
   566→              <div className="flex items-center gap-1">
   567→                {/* fix H-1 (round 4): language toggle hidden — EN translation
   568→                    is incomplete (~450 hardcoded Arabic strings). Re-enable the
   569→                    Languages button when src/messages/en.json covers all UI. */}
   570→
   571→                {/* Palette toggle (academic / modern) */}
   572→                <Button
   573→                  variant="ghost"
   574→                  size="icon"
   575→                  onClick={togglePalette}
   576→                  className="shrink-0 h-10 w-10"
   577→                  aria-label="Toggle palette"
   578→                  title={palette === "academic" ? "أكاديمي" : "عصري"}
   579→                >
   580→                  <Palette className="w-4 h-4" />
   581→                </Button>
   582→
   583→                {/* Dark / light toggle */}
   584→                <Button
   585→                  variant="ghost"
   586→                  size="icon"
   587→                  onClick={() => setTheme(theme === "dark" ? "light" : "dark")}
   588→                  className="shrink-0 h-10 w-10"
   589→                  aria-label="Toggle dark mode"
   590→                >
   591→                  {theme === "dark" ? (
   592→                    <Sun className="w-4 h-4" />
   593→                  ) : (
   594→                    <Moon className="w-4 h-4" />
   595→                  )}
   596→                </Button>
   597→
   598→                {/* Notifications bell (fix B.8 + round 10 review §3/§4):
   599→                    opens the unified notifications panel — join-request
   600→                    outcomes (students) + waiting items (supervisors) —
   601→                    with announcements still reachable inside it. */}
   602→                <Button
   603→                  variant="ghost"
   604→                  size="icon"
   605→                  onClick={() => {
   606→                    setNotifOpen(true);
   607→                    setNotifLoading(true);
   608→                    refreshNotifications().finally(() => setNotifLoading(false));
   609→                  }}
   610→                  className="shrink-0 relative h-10 w-10"
   611→                  aria-label={t("nav.notifications")}
   612→                >
   613→                  <Bell className="w-4 h-4" />
   614→                  {unreadCount + notifUnread > 0 && (
   615→                    <Badge
   616→                      variant="destructive"
   617→                      className="absolute -top-1 -right-1 h-5 min-w-5 px-1 text-xs flex items-center justify-center"
   618→                    >
   619→                      {unreadCount + notifUnread > 9 ? "9+" : unreadCount + notifUnread}
   620→                    </Badge>
   621→                  )}
   622→                </Button>
   623→
   624→                {/* Settings gear (round 26): opens the app settings screen
   625→                    — notification preferences live here now, more sections
   626→                    can be added later. Previously it only jumped to حسابي. */}
   627→                <Button
   628→                  variant="ghost"
   629→                  size="icon"
   630→                  onClick={() => navigate("SETTINGS")}
   631→                  className="shrink-0 h-10 w-10"
   632→                  aria-label="الإعدادات"
   633→                >
   634→                  <Settings className="w-4 h-4" />
   635→                </Button>
   636→              </div>
   637→            </div>
   638→
   639→            {/* Loading indicator */}
   640→            <AnimatePresence>
   641→              {isLoading && (
   642→                <motion.div
   643→                  initial={{ opacity: 0, y: -8 }}
   644→                  animate={{ opacity: 1, y: 0 }}
   645→                  exit={{ opacity: 0, y: -8 }}
   646→                  className="absolute top-full left-0 right-0 bg-primary/10 backdrop-blur-sm border-b border-primary/20 py-2 px-4 text-center"
   647→                >
   648→                  <div className="flex items-center justify-center gap-2 text-sm">
   649→                    <Loader2 className="w-3.5 h-3.5 animate-spin text-primary" />
   650→                    <span className="text-primary font-medium">
   651→                      {loadingMessage || t("common.loading")}
   652→                    </span>
   653→                  </div>
   654→                </motion.div>
   655→              )}
   656→            </AnimatePresence>
   657→          </header>
   658→        )}
   659→
   660→        {/* Main content */}
   661→        <main className="flex-1 mx-auto w-full max-w-5xl px-4 py-4 pb-24">
   662→          <AnimatePresence mode="wait">
   663→            <motion.div
   664→              key={effectiveScreen}
   665→              initial={{ opacity: 0 }}
   666→              animate={{ opacity: 1 }}
   667→              exit={{ opacity: 0 }}
   668→              transition={{ duration: 0.2 }}
   669→            >
   670→              {effectiveScreen === "HOME" && <TalibHomeScreen />}
   671→              {effectiveScreen === "COURSES" && <TalibCoursesScreen />}
   672→              {effectiveScreen === "SCHEDULE" && <TalibScheduleScreen />}
   673→              {effectiveScreen === "EXAMS" && <TalibExamsScreen />}
   674→              {effectiveScreen === "GRADES" && <TalibGradesScreen />}
   675→              {effectiveScreen === "ASSIGNMENTS" && <TalibAssignmentsScreen />}
   676→              {effectiveScreen === "FILES" && <TalibFilesScreen />}
   677→              {effectiveScreen === "TELEGRAM" && <TalibTelegramScreen />}
   678→              {effectiveScreen === "GROUP" && <TalibGroupScreen />}
   679→              {effectiveScreen === "BROWSE_GROUPS" && <TalibBrowseGroupsScreen />}
   680→              {effectiveScreen === "ANNOUNCEMENTS" && <TalibAnnouncementsScreen />}
   681→              {effectiveScreen === "COURSE_DETAIL" && (
   682→                <TalibCourseDetailScreen course={detailCourse} />
   683→              )}
   684→              {effectiveScreen === "PROFILE" && (
   685→                <TalibProfileScreen
   686→                  onSignOut={async () => {
   687→                    await signOut();
   688→                    window.history.replaceState({ talib: "HOME" }, "", "#/home");
   689→                    setCurrentScreen("HOME");
   690→                  }}
   691→                />
   692→              )}
   693→              {effectiveScreen === "SETTINGS" && <TalibSettingsScreen />}
   694→              {effectiveScreen === "ADMIN" && <TalibAdminPanelScreen />}
   695→            </motion.div>
   696→          </AnimatePresence>
   697→        </main>
   698→
   699→        {/* Bottom navigation */}
   700→        {showBottomNav && (
   701→          <TalibBottomNavBar
   702→            currentScreen={currentScreen}
   703→            onNavigate={navigate}
   704→          />
   705→        )}
   706→
   707→        <SonnerToaster position="top-center" dir={dir} />
   708→
   709→        {/* round 10 (review §3/§4): unified notifications panel */}
   710→        <TalibNotificationsSheet
   711→          open={notifOpen}
   712→          onOpenChange={setNotifOpen}
   713→          notifications={notifications}
   714→          loading={notifLoading}
   715→          announcementsUnread={unreadCount}
   716→          onMarkAllRead={markAllNotificationsRead}
   717→          onOpenAnnouncements={() => {
   718→            setNotifOpen(false);
   719→            navigate("ANNOUNCEMENTS");
   720→          }}
   721→          onItemTap={handleNotificationTap}
   722→        />
   723→      </div>
   724→    </ShellContext.Provider>
   725→  );
   726→}

   730→    <AuthProvider>
   731→      <ShellInner />
   732→    </AuthProvider>
