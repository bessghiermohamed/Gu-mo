    58→// =====================================================
    59→// fix (R12-03, P0): the app used to forget the current screen on refresh —
    60→// every reload dumped the student back on HOME, mid-task work was lost and
    61→// no screen could ever be shared/bookmarked. Each route now has a stable
    62→// hash (#/grades, #/courses, …) that is kept in sync so:
    63→//   • refresh / app re-open restores the exact screen
    64→//   • the Android/browser back gesture walks the real visit history
    65→//   • duplicate tab taps no longer poison the back stack (navigate guards)
    66→// =====================================================
    67→const ROUTE_TO_HASH: Record<ScreenRoute, string> = {
    68→  HOME: "home",
    69→  COURSES: "courses",
    70→  SCHEDULE: "schedule",
    71→  EXAMS: "exams",
    72→  TOOLS: "tools",
    73→  ASSIGNMENTS: "assignments",
    74→  FILES: "files",
    75→  GROUP: "group",
    76→  BROWSE_GROUPS: "browse-groups",
    77→  PROFILE: "profile",
    78→  SETTINGS: "settings",
    79→  ANNOUNCEMENTS: "announcements",
    80→  ADMIN: "admin",
    81→  TELEGRAM: "telegram",
    82→  COURSE_DETAIL: "course",
    83→  ONBOARDING: "onboarding", // never restored from the URL — gated below
    84→};
    85→
    86→function routeFromHash(hash: string): ScreenRoute {
    87→  const key = hash.replace(/^#\/?/, "").toLowerCase();
    88→  if (!key) return "HOME";
    89→  const found = (Object.entries(ROUTE_TO_HASH) as Array<[ScreenRoute, string]>).find(
    90→    ([, h]) => h === key
    91→  );
    92→  if (!found) return "HOME";
    93→  // never restore INTO onboarding from the URL — the onboarding gate owns it
    94→  if (found[0] === "ONBOARDING") return "HOME";
    95→  return found[0];
    96→}

   273→    }

   533→                  return;
   534→                }
   535→                setOnboardingDone(true);
   536→                window.history.replaceState({ talib: "HOME" }, "", "#/home");
   537→                setCurrentScreen("HOME");
   538→                toast.success(t("onboarding.finish"));
   539→              }}
   540→            />
   541→          </main>
   542→          <SonnerToaster position="top-center" dir={dir} />
   543→        </div>
   544→      </ShellContext.Provider>
   545→    );
   546→  }
   547→
   548→  // fix (R12-06, P0): the ADMIN panel used to render for ANY role reached
   549→  // via the notification deep-link or a typed URL. The route is now gated —
   550→  // non-supervisors land on HOME (defense in depth; the APIs already
   551→  // enforce their own authorization).
   552→  const effectiveScreen: ScreenRoute =
   553→    currentScreen === "ADMIN" && !canManageRoles(user ?? null) ? "HOME" : currentScreen;
   554→
   555→  const isAdminScreen = effectiveScreen === "ADMIN";
   556→  const showHeader = true;
   557→  const showBottomNav = !isAdminScreen;
   558→
   559→  return (
   560→    <ShellContext.Provider value={shellValue}>
   561→      <div dir={dir} className="min-h-screen bg-background flex flex-col">
   562→        {/* Header */}
   563→        {showHeader && (
   564→          <header className="sticky top-0 z-30 bg-background/95 backdrop-blur supports-[backdrop-filter]:bg-background/60 border-b border-border">
   565→            <div className="mx-auto max-w-5xl px-4 py-3 flex items-center gap-3">
   566→              {/* Back button */}
   567→              {effectiveScreen !== "HOME" && (
   568→                <Button
   569→                  variant="ghost"
   570→                  size="icon"
   571→                  onClick={navigateBack}
   572→                  className="shrink-0"
   573→                  aria-label={t("common.back")}
   574→                >
   575→                  <ChevronLeft className={cn("w-5 h-5", dir === "rtl" && "rotate-180")} />
   576→                </Button>
   577→              )}
   578→
   579→              {/* App logo + name */}
   580→              <div className="flex items-center gap-2 flex-1 min-w-0">
   581→                <img
   582→                  src="/talib/icon.svg"
   583→                  alt="Talib"
   584→                  className="w-8 h-8 shrink-0"
   585→                />
   586→                <span className="font-bold text-base truncate">
   587→                  {t("common.appName")}
   588→                </span>
   589→              </div>
   590→
   591→              {/* Action buttons */}
   592→              <div className="flex items-center gap-1">
   593→                {/* fix H-1 (round 4): language toggle hidden — EN translation
   594→                    is incomplete (~450 hardcoded Arabic strings). Re-enable the
   595→                    Languages button when src/messages/en.json covers all UI. */}
   596→
   597→                {/* Palette toggle (academic / modern) */}
   598→                <Button
   599→                  variant="ghost"
   600→                  size="icon"
   601→                  onClick={togglePalette}
   602→                  className="shrink-0 h-10 w-10"
   603→                  aria-label="Toggle palette"
   604→                  title={palette === "academic" ? "أكاديمي" : "عصري"}
   605→                >
   606→                  <Palette className="w-4 h-4" />
   607→                </Button>
   608→
   609→                {/* Dark / light toggle */}
   610→                <Button
   611→                  variant="ghost"
   612→                  size="icon"
   613→                  onClick={() => setTheme(theme === "dark" ? "light" : "dark")}
   614→                  className="shrink-0 h-10 w-10"
   615→                  aria-label="Toggle dark mode"
   616→                >
   617→                  {theme === "dark" ? (
   618→                    <Sun className="w-4 h-4" />
   619→                  ) : (
   620→                    <Moon className="w-4 h-4" />
   621→                  )}
   622→                </Button>
   623→
   624→                {/* Notifications bell (fix B.8 + round 10 review §3/§4):
   625→                    opens the unified notifications panel — join-request
   626→                    outcomes (students) + waiting items (supervisors) —
   627→                    with announcements still reachable inside it. */}
   628→                <Button
   629→                  variant="ghost"
   630→                  size="icon"
   631→                  onClick={() => {
   632→                    setNotifOpen(true);
   633→                    setNotifLoading(true);
   634→                    refreshNotifications().finally(() => setNotifLoading(false));
   635→                  }}
   636→                  className="shrink-0 relative h-10 w-10"
   637→                  aria-label={t("nav.notifications")}
   638→                >
   639→                  <Bell className="w-4 h-4" />
   640→                  {unreadCount + notifUnread > 0 && (
   641→                    <Badge
   642→                      variant="destructive"
   643→                      className="absolute -top-1 -right-1 h-5 min-w-5 px-1 text-xs flex items-center justify-center"
   644→                    >
   645→                      {unreadCount + notifUnread > 9 ? "9+" : unreadCount + notifUnread}
   646→                    </Badge>
   647→                  )}
   648→                </Button>
   649→
   650→                {/* Settings gear (round 26): opens the app settings screen
   651→                    — notification preferences live here now, more sections
   652→                    can be added later. Previously it only jumped to حسابي.
   653→                    id used by the first-run tour (review §15). */}
   654→                <Button
   655→                  id="talib-tour-gear"
   656→                  variant="ghost"
   657→                  size="icon"
   658→                  onClick={() => navigate("SETTINGS")}
   659→                  className="shrink-0 h-10 w-10"
   660→                  aria-label="الإعدادات"
   661→                >
   662→                  <Settings className="w-4 h-4" />
   663→                </Button>
   664→              </div>
   665→            </div>
   666→
   667→            {/* Loading indicator */}
   668→            <AnimatePresence>
   669→              {isLoading && (
   670→                <motion.div
   671→                  initial={{ opacity: 0, y: -8 }}
   672→                  animate={{ opacity: 1, y: 0 }}
   673→                  exit={{ opacity: 0, y: -8 }}
   674→                  className="absolute top-full left-0 right-0 bg-primary/10 backdrop-blur-sm border-b border-primary/20 py-2 px-4 text-center"
   675→                >
   676→                  <div className="flex items-center justify-center gap-2 text-sm">
   677→                    <Loader2 className="w-3.5 h-3.5 animate-spin text-primary" />
   678→                    <span className="text-primary font-medium">
   679→                      {loadingMessage || t("common.loading")}
   680→                    </span>
   681→                  </div>
   682→                </motion.div>
   683→              )}
   684→            </AnimatePresence>
   685→          </header>
   686→        )}
   687→
   688→        {/* Main content */}
   689→        <main className="flex-1 mx-auto w-full max-w-5xl px-4 py-4 pb-24">
   690→          <AnimatePresence mode="wait">
   691→            <motion.div
   692→              key={effectiveScreen}
   693→              initial={{ opacity: 0 }}
   694→              animate={{ opacity: 1 }}
   695→              exit={{ opacity: 0 }}
   696→              transition={{ duration: 0.2 }}
   697→            >
   698→              {effectiveScreen === "HOME" && <TalibHomeScreen />}
   699→              {effectiveScreen === "COURSES" && <TalibCoursesScreen />}
   700→              {effectiveScreen === "SCHEDULE" && <TalibScheduleScreen />}
   701→              {effectiveScreen === "EXAMS" && <TalibExamsScreen />}
   702→              {/* round 31: tools got their own standalone screen — the TOOLS
   703→                  tile no longer opens the files screen on a pre-selected tab
   704→                  (owner feedback: the files/tools split had duplicated
   705→                  navigation — tools were both a tile and a files tab). */}
   706→              {effectiveScreen === "TOOLS" && <TalibToolsScreen />}
   707→              {effectiveScreen === "ASSIGNMENTS" && <TalibAssignmentsScreen />}
   708→              {effectiveScreen === "FILES" && <TalibFilesScreen />} {/* tabs: المكتبة / ملاحظاتي / سحابتي */}
   709→              {effectiveScreen === "TELEGRAM" && <TalibTelegramScreen />}
   710→              {effectiveScreen === "GROUP" && <TalibGroupScreen />}
   711→              {effectiveScreen === "BROWSE_GROUPS" && <TalibBrowseGroupsScreen />}
   712→              {effectiveScreen === "ANNOUNCEMENTS" && <TalibAnnouncementsScreen />}
   713→              {effectiveScreen === "COURSE_DETAIL" && (
   714→                <TalibCourseDetailScreen course={detailCourse} />
   715→              )}
   716→              {effectiveScreen === "PROFILE" && (
   717→                <TalibProfileScreen
   718→                  onSignOut={async () => {
   719→                    await signOut();
   720→                    window.history.replaceState({ talib: "HOME" }, "", "#/home");
   721→                    setCurrentScreen("HOME");
   722→                  }}
   723→                />
   724→              )}
   725→              {effectiveScreen === "SETTINGS" && <TalibSettingsScreen />}
   726→              {effectiveScreen === "ADMIN" && <TalibAdminPanelScreen />}
   727→            </motion.div>
   728→          </AnimatePresence>
   729→        </main>
   730→
   731→        {/* Bottom navigation */}
   732→        {showBottomNav && (
   733→          <TalibBottomNavBar
   734→            currentScreen={currentScreen}
   735→            onNavigate={navigate}
   736→          />
   737→        )}
   738→
   739→        <SonnerToaster position="top-center" dir={dir} />
   740→
   741→        {/* round 27 (review §15): first-run tour — three steps (services
   742→            grid → gear → حسابي), shown once, dismissible at every step. */}
   743→        <TalibTourOverlay currentScreen={currentScreen} />
   744→
   745→        {/* round 10 (review §3/§4): unified notifications panel */}
   746→        <TalibNotificationsSheet
   747→          open={notifOpen}
   748→          onOpenChange={setNotifOpen}
   749→          notifications={notifications}
   750→          loading={notifLoading}
   751→          announcementsUnread={unreadCount}
   752→          onMarkAllRead={markAllNotificationsRead}
   753→          onOpenAnnouncements={() => {
   754→            setNotifOpen(false);
   755→            navigate("ANNOUNCEMENTS");
   756→          }}
   757→          onItemTap={handleNotificationTap}
   758→        />
   759→      </div>
   760→    </ShellContext.Provider>
   761→  );
   762→}
