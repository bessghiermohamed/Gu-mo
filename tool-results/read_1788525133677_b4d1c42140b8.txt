    55→    55→// =====================================================
    56→    56→// fix (R12-03, P0): the app used to forget the current screen on refresh —
    57→    57→// every reload dumped the student back on HOME, mid-task work was lost and
    58→    58→// no screen could ever be shared/bookmarked. Each route now has a stable
    59→    59→// hash (#/grades, #/courses, …) that is kept in sync so:
    60→    60→//   • refresh / app re-open restores the exact screen
    61→    61→//   • the Android/browser back gesture walks the real visit history
    62→    62→//   • duplicate tab taps no longer poison the back stack (navigate guards)
    63→    63→// =====================================================
    64→    64→const ROUTE_TO_HASH: Record<ScreenRoute, string> = {
    65→    65→  HOME: "home",
    66→    66→  COURSES: "courses",
    67→    67→  SCHEDULE: "schedule",
    68→    68→  EXAMS: "exams",
    69→    69→  GRADES: "grades",
    70→    70→  ASSIGNMENTS: "assignments",
    71→    71→  FILES: "files",
    72→    72→  GROUP: "group",
    73→    73→  BROWSE_GROUPS: "browse-groups",
    74→    74→  PROFILE: "profile",
    75→    75→  ANNOUNCEMENTS: "announcements",
    76→    76→  ADMIN: "admin",
    77→    77→  TELEGRAM: "telegram",
    78→    78→  COURSE_DETAIL: "course",
    79→    79→  ONBOARDING: "onboarding", // never restored from the URL — gated below
    80→    80→};
    81→    81→
    82→    82→function routeFromHash(hash: string): ScreenRoute {
    83→    83→  const key = hash.replace(/^#\/?/, "").toLowerCase();
    84→    84→  if (!key) return "HOME";
    85→    85→  const found = (Object.entries(ROUTE_TO_HASH) as Array<[ScreenRoute, string]>).find(
    86→    86→    ([, h]) => h === key
    87→    87→  );
    88→    88→  if (!found) return "HOME";
    89→    89→  // never restore INTO onboarding from the URL — the onboarding gate owns it
    90→    90→  if (found[0] === "ONBOARDING") return "HOME";
    91→    91→  return found[0];
    92→    92→}
