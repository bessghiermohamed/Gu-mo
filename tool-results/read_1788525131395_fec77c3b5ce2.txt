    55→// =====================================================
    56→// fix (R12-03, P0): the app used to forget the current screen on refresh —
    57→// every reload dumped the student back on HOME, mid-task work was lost and
    58→// no screen could ever be shared/bookmarked. Each route now has a stable
    59→// hash (#/grades, #/courses, …) that is kept in sync so:
    60→//   • refresh / app re-open restores the exact screen
    61→//   • the Android/browser back gesture walks the real visit history
    62→//   • duplicate tab taps no longer poison the back stack (navigate guards)
    63→// =====================================================
    64→const ROUTE_TO_HASH: Record<ScreenRoute, string> = {
    65→  HOME: "home",
    66→  COURSES: "courses",
    67→  SCHEDULE: "schedule",
    68→  EXAMS: "exams",
    69→  GRADES: "grades",
    70→  ASSIGNMENTS: "assignments",
    71→  FILES: "files",
    72→  GROUP: "group",
    73→  BROWSE_GROUPS: "browse-groups",
    74→  PROFILE: "profile",
    75→  ANNOUNCEMENTS: "announcements",
    76→  ADMIN: "admin",
    77→  TELEGRAM: "telegram",
    78→  COURSE_DETAIL: "course",
    79→  ONBOARDING: "onboarding", // never restored from the URL — gated below
    80→};
    81→
    82→function routeFromHash(hash: string): ScreenRoute {
    83→  const key = hash.replace(/^#\/?/, "").toLowerCase();
    84→  if (!key) return "HOME";
    85→  const found = (Object.entries(ROUTE_TO_HASH) as Array<[ScreenRoute, string]>).find(
    86→    ([, h]) => h === key
    87→  );
    88→  if (!found) return "HOME";
    89→  // never restore INTO onboarding from the URL — the onboarding gate owns it
    90→  if (found[0] === "ONBOARDING") return "HOME";
    91→  return found[0];
    92→}

   141→  const [notifLoading, setNotifLoading] = React.useState(false);
   142→  const [mounted, setMounted] = React.useState(false);
   143→  const [onboardingDone, setOnboardingDone] = React.useState(true);
   144→
   145→  React.useEffect(() => {
   146→    setMounted(true);
   147→  }, []);
   148→
   149→  // Check if user needs onboarding
   150→  // fix (R25): onboarding used to be gated by localStorage ONLY — a
   151→  // fully-registered student who cleared browser data, switched devices,
   152→  // or reinstalled was forced through the whole setup flow again even
   153→  // though their profile exists in the DB. The DB row is now the source
   154→  // of truth: /api/onboarding/complete (and role grants) persist the
   155→  // academic scope on the user row, so a non-null scope specialty/year
   156→  // means the profile is complete and the flow is skipped — with an
   157→  // EMPTY localStorage too. The localStorage flag stays as a fast local
   158→  // cache and is backfilled after a DB-confirmed skip. (Note:
   159→  // assignedSpecialtyId is NOT a valid marker — signup pre-fills it.)
   160→  React.useEffect(() => {
   161→    if (!user) return;
   162→    const flagKey = `talib-onboarding-${user.id}`;
   163→    // DB-backed: scope_academic_year_id / scope_specialty_id are null at
   164→    // signup and only written by onboarding completion (or a role grant).
   165→    const hasDbScope =
   166→      user.scopeAcademicYearId != null || user.scopeSpecialtyId != null;
   167→    const localDone = localStorage.getItem(flagKey) === "true";
   168→    const done = localDone || hasDbScope;
   169→    setOnboardingDone(done);
   170→    if (!done) {
   171→      setCurrentScreen("ONBOARDING");
   172→    } else if (hasDbScope && !localDone) {
   173→      // cache the DB verdict so subsequent loads skip the check instantly
   174→      try {
   175→        localStorage.setItem(flagKey, "true");
   176→      } catch {
   177→        // private mode — the DB check simply re-runs next load
   178→      }
   179→    }
   180→  }, [user]);
   181→
   182→  // Fetch unread count (fix B.8)
   183→  const refreshUnreadCount = React.useCallback(async () => {
   184→    if (!user) {
   185→      setUnreadCount(0);
   186→      return;
   187→    }
   188→    try {
   189→      const res = await fetch("/api/announcements/unread-count", {
   190→        cache: "no-store",
   191→      });
   192→      if (res.ok) {
   193→        const data = await res.json();
   194→        setUnreadCount(data.count ?? 0);
   195→      }
   196→    } catch {
   197→      // silent fail
   198→    }
   199→  }, [user]);
   200→
   201→  // round 10 (review §3/§4): app notifications feed — join-request
   202→  // outcomes for students, waiting items for supervisors.
   203→  const refreshNotifications = React.useCallback(async () => {
   204→    if (!user) {
   205→      setNotifications([]);
   206→      setNotifUnread(0);
   207→      return;
   208→    }
   209→    try {
   210→      const res = await fetch("/api/notifications", { cache: "no-store" });
   211→      if (res.ok) {
   212→        const data = await res.json();
   213→        setNotifications(data.notifications ?? []);
   214→        setNotifUnread(data.unreadCount ?? 0);
   215→      }
   216→    } catch {
   217→      // silent fail — badge simply stays stale until next poll
   218→    }
   219→  }, [user]);
   220→
   221→  React.useEffect(() => {
   222→    refreshUnreadCount();
   223→    refreshNotifications();
   224→    const interval = setInterval(() => {
   225→      refreshUnreadCount();
   226→      refreshNotifications();

   228→    return () => clearInterval(interval);
   229→  }, [refreshUnreadCount, refreshNotifications]);
   230→
   231→  // fix (R12): the announcements screen marks its items read — the badge
   232→  // refreshes IMMEDIATELY instead of waiting for the next 30s poll.
   233→  React.useEffect(() => {
   234→    const onAnnRead = () => { refreshUnreadCount(); };
   235→    window.addEventListener("talib-ann-read", onAnnRead);
   236→    return () => window.removeEventListener("talib-ann-read", onAnnRead);
   237→  }, [refreshUnreadCount]);
   238→
   239→  const markAllNotificationsRead = React.useCallback(async () => {
   240→    try {
   241→      await fetch("/api/notifications/read", {
   242→        method: "POST",
   243→        headers: { "Content-Type": "application/json" },
   244→        body: JSON.stringify({ all: true }),
   245→      });
   246→      setNotifUnread(0);
   247→      setNotifications((prev) => prev.map((n) => ({ ...n, readAt: n.readAt ?? new Date().toISOString() })));
   248→    } catch {
   249→      // silent
   250→    }
   251→  }, []);
   252→
   253→  const navigate = React.useCallback((route: ScreenRoute) => {
   254→    if (route === "COURSE_DETAIL") return; // use navigateToCourse
   255→    setCurrentScreen((prev) => {
   256→      // fix (R12-03): tapping the tab you are ALREADY on used to push a
   257→      // duplicate history entry, so the back chevron appeared to do nothing.
   258→      if (prev === route) return prev;
   259→      historyStackRef.current.push(prev);
   260→      return route;
   261→    });

   263→    if (window.location.hash !== target) {
   264→      window.history.pushState({ talib: route }, "", target);

   266→  }, []);
   267→
   268→  const navigateToCourse = React.useCallback((course: CourseSummary) => {

   270→    try {
   271→      sessionStorage.setItem("talib-course", JSON.stringify(course));
   272→    } catch {
   273→      // storage disabled — in-session navigation still works
   274→    }
   275→    setCurrentScreen((prev) => {
   276→      if (prev === "COURSE_DETAIL") return prev;
   277→      historyStackRef.current.push(prev);
   278→      return "COURSE_DETAIL";
   279→    });
   280→    const target = "#/course";
   281→    if (window.location.hash !== target) {
   282→      window.history.pushState({ talib: "COURSE_DETAIL" }, "", target);
   283→    }
   284→  }, []);
   285→
   286→  const navigateBack = React.useCallback(() => {
   287→    const stack = historyStackRef.current;
   288→    if (stack.length > 0) {
   289→      const prev = stack.pop()!;

   291→      if (prev !== "COURSE_DETAIL") setDetailCourse(null);
   292→      const target = `#/${ROUTE_TO_HASH[prev]}`;
   293→      if (window.location.hash !== target) {
   294→        window.history.pushState({ talib: prev }, "", target);
   295→      }
   296→    } else {
   297→      setCurrentScreen("HOME");
   298→      const target = "#/home";
   299→      if (window.location.hash !== target) {
   300→        window.history.replaceState({ talib: "HOME" }, "", target);
   301→      }
   302→    }
   303→  }, []);
   304→
   305→  // Browser / Android back gesture: the previous hash wins. The in-app stack
   306→  // is cleared (the OS already walked back for us); the next in-app back
   307→  // then safely falls home.

   309→    function onPopState() {
   310→      historyStackRef.current = [];
   311→      const route = routeFromHash(window.location.hash);
   312→      setCurrentScreen((prev) => (prev === route ? prev : route));
   313→      if (route === "COURSE_DETAIL") {
   314→        try {
   315→          const raw = sessionStorage.getItem("talib-course");
   316→          if (raw) setDetailCourse(JSON.parse(raw) as CourseSummary);
   317→        } catch {
   318→          // fall through — render guard below handles a missing course
   319→        }
   320→      }
   321→    }
   322→    window.addEventListener("popstate", onPopState);
   323→    return () => window.removeEventListener("popstate", onPopState);
   324→  }, []);
   325→
   326→  // Restore the last screen after a refresh / from a shared deep link once
   327→  // we know who the user is (needed for the ADMIN role guard).
   328→  React.useEffect(() => {
   329→    if (!user) return;
   330→    if (!window.location.hash) {
   331→      window.history.replaceState({ talib: "HOME" }, "", "#/home");
   332→      return;
   333→    }
   334→    const route = routeFromHash(window.location.hash);
   335→    if (route === "COURSE_DETAIL") {
   336→      try {
   337→        const raw = sessionStorage.getItem("talib-course");
   338→        if (raw) {
   339→          setDetailCourse(JSON.parse(raw) as CourseSummary);
   340→          setCurrentScreen("COURSE_DETAIL");
   341→          return;
   342→        }
   343→      } catch {
   344→        // corrupted session storage — fall through to HOME
   345→      }
   346→      window.history.replaceState({ talib: "HOME" }, "", "#/home");
   347→      setCurrentScreen("HOME");
   348→      return;
   349→    }
   350→    setCurrentScreen(route);
   351→  }, [user]);
   352→
   353→  const handleNotificationTap = React.useCallback(
   354→    (item: AppNotificationItem) => {
   355→      // mark the single notification read
   356→      if (item.readAt == null) {
   357→        fetch("/api/notifications/read", {
   358→          method: "POST",
   359→          headers: { "Content-Type": "application/json" },
   360→          body: JSON.stringify({ ids: [item.id] }),
   361→        })
   362→          .then(() => {
   363→            setNotifUnread((c) => Math.max(0, c - 1));
   364→            setNotifications((prev) =>
   365→              prev.map((n) => (n.id === item.id ? { ...n, readAt: new Date().toISOString() } : n))
   366→            );
   367→          })
   368→          .catch(() => {});
   369→      }
   370→      // deep link by type (review §4: the notification takes you to the
   371→      // place where you can act on it)
   372→      setNotifOpen(false);
   373→      if (item.type === "join_approved" || item.type === "join_rejected") {
   374→        navigate("BROWSE_GROUPS");
   375→      } else if (item.type === "join_new" || item.type === "report_new") {
   376→        navigate("ADMIN");
   377→      }
   378→    },
   379→    [navigate]
   380→  );
   381→
   382→  const showLoading = React.useCallback((message?: string) => {
   383→    setIsLoading(true);
   384→    setLoadingMessage(message ?? "");
   385→  }, []);
   386→
   387→  const hideLoading = React.useCallback(() => {
   388→    setIsLoading(false);
   389→    setLoadingMessage("");
   390→  }, []);
   391→
   392→  const shellValue: ShellContextValue = {
   393→    currentScreen,
   394→    navigate,
   395→    navigateBack,
   396→    showLoading,
   397→    hideLoading,
   398→    detailCourse,
   399→    navigateToCourse,
   400→  };
   401→
   402→  // Auth loading state
   403→  if (authLoading || !mounted) {
   404→    return (
   405→      <div className="flex items-center justify-center min-h-screen bg-background">
   406→        <div className="flex flex-col items-center gap-3">
   407→          <div className="relative w-16 h-16">
   408→            <img
   409→              src="/talib/icon.svg"
   410→              alt="Talib"
   411→              className="w-full h-full object-contain"
   412→            />
   413→          </div>
   414→          <div className="flex items-center gap-2 text-muted-foreground text-sm">
   415→            <Loader2 className="w-4 h-4 animate-spin" />
   416→            <span>{t("common.loading")}</span>
   417→          </div>
   418→        </div>
   419→      </div>
   420→    );
   421→  }
   422→
   423→  // Not authenticated → show login
   424→  if (!user) {
   425→    return (
   426→      <div dir={dir} className="min-h-screen bg-background">
   427→        <TalibLoginScreen />
   428→        <SonnerToaster position="top-center" dir={dir} />
   429→      </div>
   430→    );
   431→  }
   432→
   433→  // Needs onboarding — pure render gate (no setState during render):
   434→  // onboarding wins over ANY screen, including a restored deep link.
   435→  if (!onboardingDone) {
   436→    return (
   437→      <ShellContext.Provider value={shellValue}>
   438→        <div dir={dir} className="min-h-screen bg-background flex flex-col">
   439→          <main className="flex-1">
   440→            <TalibOnboardingScreen
   441→              onComplete={async () => {
   442→                if (user) {
   443→                  localStorage.setItem(
   444→                    `talib-onboarding-${user.id}`,
   445→                    "true"
   446→                  );
   447→                }
   448→                // fix أ.3/أ.4 (round 3): the session user used to stay stale
   449→                // after onboarding (assignedSpecialtyId = first specialty,
   450→                // null year/track) so the join screen leaked other
   451→                // specialties' groups until a full page reload. Refresh the
   452→                // session user BEFORE navigating home.
   453→                await refresh();
   454→                setOnboardingDone(true);
   455→                window.history.replaceState({ talib: "HOME" }, "", "#/home");
   456→                setCurrentScreen("HOME");
   457→                toast.success(t("onboarding.finish"));
   458→              }}
   459→            />
   460→          </main>
   461→          <SonnerToaster position="top-center" dir={dir} />

   463→      </ShellContext.Provider>
   464→    );
   465→  }
   466→
   467→  // fix (R12-06, P0): the ADMIN panel used to render for ANY role reached
   468→  // via the notification deep-link or a typed URL. The route is now gated —
   469→  // non-supervisors land on HOME (defense in depth; the APIs already
   470→  // enforce their own authorization).
   471→  const effectiveScreen: ScreenRoute =
   472→    currentScreen === "ADMIN" && !canManageRoles(user ?? null) ? "HOME" : currentScreen;
   473→
   474→  const isAdminScreen = effectiveScreen === "ADMIN";
   475→  const showHeader = true;
   476→  const showBottomNav = !isAdminScreen;
   477→
   478→  return (
   479→    <ShellContext.Provider value={shellValue}>
   480→      <div dir={dir} className="min-h-screen bg-background flex flex-col">
   481→        {/* Header */}
   482→        {showHeader && (
   483→          <header className="sticky top-0 z-30 bg-background/95 backdrop-blur supports-[backdrop-filter]:bg-background/60 border-b border-border">
   484→            <div className="mx-auto max-w-5xl px-4 py-3 flex items-center gap-3">

   486→              {effectiveScreen !== "HOME" && (
   487→                <Button
   488→                  variant="ghost"
   489→                  size="icon"
   490→                  onClick={navigateBack}
   491→                  className="shrink-0"
   492→                  aria-label={t("common.back")}
   493→                >
   494→                  <ChevronLeft className={cn("w-5 h-5", dir === "rtl" && "rotate-180")} />
   495→                </Button>
   496→              )}
   497→
   498→              {/* App logo + name */}
   499→              <div className="flex items-center gap-2 flex-1 min-w-0">
   500→                <img
   501→                  src="/talib/icon.svg"
   502→                  alt="Talib"
   503→                  className="w-8 h-8 shrink-0"
   504→                />
   505→                <span className="font-bold text-base truncate">
   506→                  {t("common.appName")}
   507→                </span>
   508→              </div>
   509→
   510→              {/* Action buttons */}
   511→              <div className="flex items-center gap-1">
   512→                {/* fix H-1 (round 4): language toggle hidden — EN translation
   513→                    is incomplete (~450 hardcoded Arabic strings). Re-enable the
   514→                    Languages button when src/messages/en.json covers all UI. */}
   515→
   516→                {/* Palette toggle (academic / modern) */}
   517→                <Button
   518→                  variant="ghost"
   519→                  size="icon"
   520→                  onClick={togglePalette}
   521→                  className="shrink-0 h-10 w-10"
   522→                  aria-label="Toggle palette"
   523→                  title={palette === "academic" ? "أكاديمي" : "عصري"}
   524→                >
   525→                  <Palette className="w-4 h-4" />
   526→                </Button>
   527→
   528→                {/* Dark / light toggle */}
   529→                <Button
   530→                  variant="ghost"
   531→                  size="icon"
   532→                  onClick={() => setTheme(theme === "dark" ? "light" : "dark")}
   533→                  className="shrink-0 h-10 w-10"
   534→                  aria-label="Toggle dark mode"
   535→                >
   536→                  {theme === "dark" ? (
   537→                    <Sun className="w-4 h-4" />

   542→
   543→                {/* Notifications bell (fix B.8 + round 10 review §3/§4):

   546→                    with announcements still reachable inside it. */}
   547→                <Button
   548→                  variant="ghost"
   549→                  size="icon"
   550→                  onClick={() => {
   551→                    setNotifOpen(true);
   552→                    setNotifLoading(true);
   553→                    refreshNotifications().finally(() => setNotifLoading(false));
   554→                  }}
   555→                  className="shrink-0 relative h-10 w-10"
   556→                  aria-label={t("nav.notifications")}
   557→                >
   558→                  <Bell className="w-4 h-4" />
   559→                  {unreadCount + notifUnread > 0 && (
   560→                    <Badge
   561→                      variant="destructive"
   562→                      className="absolute -top-1 -right-1 h-5 min-w-5 px-1 text-xs flex items-center justify-center"
   563→                    >
   564→                      {unreadCount + notifUnread > 9 ? "9+" : unreadCount + notifUnread}
   565→                    </Badge>
   566→                  )}
   567→                </Button>
   568→
   569→                {/* Admin / Profile menu */}
   570→                <Button
   571→                  variant="ghost"
   572→                  size="icon"
   573→                  onClick={() => navigate("PROFILE")}
   574→                  className="shrink-0 h-10 w-10"
   575→                  aria-label={t("nav.profile")}
   576→                >
   577→                  <Settings className="w-4 h-4" />
   578→                </Button>
   579→              </div>
   580→            </div>
   581→
   582→            {/* Loading indicator */}
   583→            <AnimatePresence>
   584→              {isLoading && (
   585→                <motion.div
   586→                  initial={{ opacity: 0, y: -8 }}
   587→                  animate={{ opacity: 1, y: 0 }}
   588→                  exit={{ opacity: 0, y: -8 }}
   589→                  className="absolute top-full left-0 right-0 bg-primary/10 backdrop-blur-sm border-b border-primary/20 py-2 px-4 text-center"
   590→                >
   591→                  <div className="flex items-center justify-center gap-2 text-sm">
   592→                    <Loader2 className="w-3.5 h-3.5 animate-spin text-primary" />
   593→                    <span className="text-primary font-medium">
   594→                      {loadingMessage || t("common.loading")}
   595→                    </span>
   596→                  </div>
   597→                </motion.div>
   598→              )}
   599→            </AnimatePresence>
   600→          </header>
   601→        )}
   602→
   603→        {/* Main content */}
   604→        <main className="flex-1 mx-auto w-full max-w-5xl px-4 py-4 pb-24">
   605→          <AnimatePresence mode="wait">
   606→            <motion.div
   607→              key={effectiveScreen}
   608→              initial={{ opacity: 0 }}
   609→              animate={{ opacity: 1 }}
   610→              exit={{ opacity: 0 }}
   611→              transition={{ duration: 0.2 }}
   612→            >
   613→              {effectiveScreen === "HOME" && <TalibHomeScreen />}
   614→              {effectiveScreen === "COURSES" && <TalibCoursesScreen />}
   615→              {effectiveScreen === "SCHEDULE" && <TalibScheduleScreen />}
   616→              {effectiveScreen === "EXAMS" && <TalibExamsScreen />}
   617→              {effectiveScreen === "GRADES" && <TalibGradesScreen />}
   618→              {effectiveScreen === "ASSIGNMENTS" && <TalibAssignmentsScreen />}
   619→              {effectiveScreen === "FILES" && <TalibFilesScreen />}
   620→              {effectiveScreen === "TELEGRAM" && <TalibTelegramScreen />}
   621→              {effectiveScreen === "GROUP" && <TalibGroupScreen />}
   622→              {effectiveScreen === "BROWSE_GROUPS" && <TalibBrowseGroupsScreen />}
   623→              {effectiveScreen === "ANNOUNCEMENTS" && <TalibAnnouncementsScreen />}
   624→              {effectiveScreen === "COURSE_DETAIL" && (
   625→                <TalibCourseDetailScreen course={detailCourse} />
   626→              )}
   627→              {effectiveScreen === "PROFILE" && (
   628→                <TalibProfileScreen
   629→                  onSignOut={async () => {
   630→                    await signOut();
   631→                    window.history.replaceState({ talib: "HOME" }, "", "#/home");
   632→                    setCurrentScreen("HOME");
   633→                  }}
   634→                />
   635→              )}
   636→              {effectiveScreen === "ADMIN" && <TalibAdminPanelScreen />}
   637→            </motion.div>
   638→          </AnimatePresence>
   639→        </main>
   640→
   641→        {/* Bottom navigation */}
   642→        {showBottomNav && (
   643→          <TalibBottomNavBar
   644→            currentScreen={currentScreen}
   645→            onNavigate={navigate}
   646→          />
   647→        )}
   648→
   649→        <SonnerToaster position="top-center" dir={dir} />
   650→
   651→        {/* round 10 (review §3/§4): unified notifications panel */}
   652→        <TalibNotificationsSheet
   653→          open={notifOpen}
   654→          onOpenChange={setNotifOpen}
   655→          notifications={notifications}
   656→          loading={notifLoading}
   657→          announcementsUnread={unreadCount}
   658→          onMarkAllRead={markAllNotificationsRead}
   659→          onOpenAnnouncements={() => {
   660→            setNotifOpen(false);
   661→            navigate("ANNOUNCEMENTS");
   662→          }}
   663→          onItemTap={handleNotificationTap}
   664→        />
   665→      </div>
   666→    </ShellContext.Provider>
   667→  );
   668→}

   672→    <AuthProvider>
