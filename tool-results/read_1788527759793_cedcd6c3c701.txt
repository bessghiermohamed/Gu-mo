   256→   256→function AddSlotDialog({
   257→   257→  open,
   258→   258→  onOpenChange,
   259→   259→  onCreated,
   260→   260→}: {
   261→   261→  open: boolean;
   262→   262→  onOpenChange: (v: boolean) => void;
   263→   263→  onCreated: () => void;
   264→   264→}) {
   265→   265→  const { t } = useI18n();
   266→   266→  const [dayOfWeek, setDayOfWeek] = React.useState("1");
   267→   267→  const [startTime, setStartTime] = React.useState("");
   268→   268→  const [endTime, setEndTime] = React.useState("");
   269→   269→  const [moduleName, setModuleName] = React.useState("");
   270→   270→  const [type, setType] = React.useState("محاضرة");
   271→   271→  const [room, setRoom] = React.useState("");
   272→   272→  const [professor, setProfessor] = React.useState("");
   273→   273→  const [saving, setSaving] = React.useState(false);
   274→   274→
   275→   275→  async function handleSave() {
   276→   276→    if (!startTime.trim() || !moduleName.trim()) {
   277→   277→      toast.error("وقت البداية واسم المقياس مطلوبان");
   278→   278→      return;
   279→   279→    }
   280→   280→    setSaving(true);
   281→   281→    try {
   282→   282→      const res = await fetch("/api/schedule", {
   283→   283→        method: "POST",
   284→   284→        headers: { "Content-Type": "application/json" },
   285→   285→        body: JSON.stringify({
   286→   286→          dayOfWeek: parseInt(dayOfWeek),
   287→   287→          startTime: startTime.trim(),
   288→   288→          endTime: endTime.trim(),
   289→   289→          moduleName: moduleName.trim(),
   290→   290→          type,
   291→   291→          room: room.trim(),
   292→   292→          professor: professor.trim(),
   293→   293→        }),
   294→   294→      });
   295→   295→      const data = await res.json();
   296→   296→      if (!res.ok) {
   297→   297→        toast.error(data.error ?? "فشل الحفظ");
   298→   298→        return;
   299→   299→      }
   300→   300→      toast.success("تمت إضافة الحصة بنجاح");
   301→   301→      onOpenChange(false);
   302→   302→      setStartTime("");
   303→   303→      setEndTime("");
   304→   304→      setModuleName("");
   305→   305→      setRoom("");
   306→   306→      setProfessor("");
   307→   307→      onCreated();
   308→   308→    } finally {
   309→   309→      setSaving(false);
   310→   310→    }
   311→   311→  }
   312→   312→
   313→   313→  return (
   314→   314→    <Dialog open={open} onOpenChange={onOpenChange}>
   315→   315→      <DialogTrigger asChild>
   316→   316→        <Button className="w-full">
   317→   317→          <Plus className="w-4 h-4 ml-2" />
   318→   318→          {t("schedule.addSlot")}
   319→   319→        </Button>
   320→   320→      </DialogTrigger>
   321→   321→      <DialogContent>
   322→   322→        <DialogHeader>
   323→   323→          <DialogTitle>إضافة حصة جديدة</DialogTitle>
   324→   324→        </DialogHeader>
   325→   325→        <div className="space-y-3 py-2">
   326→   326→          <div className="space-y-1.5">
   327→   327→            <Label>اليوم</Label>
   328→   328→            <select
   329→   329→              value={dayOfWeek}
   330→   330→              onChange={(e) => setDayOfWeek(e.target.value)}
   331→   331→              className="flex h-9 w-full rounded-md border border-input bg-transparent px-3 py-1 text-sm"
   332→   332→            >
   333→   333→              <option value="1">الأحد</option>
   334→   334→              <option value="2">الإثنين</option>
   335→   335→              <option value="3">الثلاثاء</option>
   336→   336→              <option value="4">الأربعاء</option>
   337→   337→              <option value="5">الخميس</option>
   338→   338→            </select>
   339→   339→          </div>
   340→   340→          <div className="grid grid-cols-2 gap-2">
   341→   341→            <div className="space-y-1.5">
   342→   342→              <Label>وقت البداية</Label>
   343→   343→              <Input
   344→   344→                type="time"
   345→   345→                value={startTime}
   346→   346→                onChange={(e) => setStartTime(e.target.value)}
   347→   347→              />
   348→   348→            </div>
   349→   349→            <div className="space-y-1.5">
   350→   350→              <Label>وقت النهاية</Label>
   351→   351→              <Input
   352→   352→                type="time"
   353→   353→                value={endTime}
   354→   354→                onChange={(e) => setEndTime(e.target.value)}
   355→   355→              />
   356→   356→            </div>
   357→   357→          </div>

   360→   360→            <Input
   361→   361→              value={moduleName}
   362→   362→              onChange={(e) => setModuleName(e.target.value)}
   363→   363→              placeholder="مثال: الأدب الجاهلي"
   364→   364→            />

   380→   380→              <Label>القاعة</Label>

   382→   382→                value={room}

   385→   385→              />
   386→   386→            </div>
   387→   387→            <div className="space-y-1.5">

   389→   389→              <Input
   390→   390→                value={professor}
   391→   391→                onChange={(e) => setProfessor(e.target.value)}

   393→   393→              />
   394→   394→            </div>
   395→   395→          </div>

   402→   402→            {saving && <Loader2 className="w-4 h-4 ml-1 animate-spin" />}
   403→   403→            حفظ الحصة
   404→   404→          </Button>
   405→   405→        </DialogFooter>

   407→   407→    </Dialog>

   411→   411→function ImageSchedule() {
   412→   412→  const { t } = useI18n();
   413→   413→  const [imageUrl, setImageUrl] = React.useState<string | null>(null);
   414→   414→
   415→   415→  function handleUpload(e: React.ChangeEvent<HTMLInputElement>) {
   416→   416→    const file = e.target.files?.[0];
   417→   417→    if (!file) return;
   418→   418→    if (!["image/png", "image/jpeg", "application/pdf"].includes(file.type)) {
   419→   419→      toast.error("صيغة غير مدعومة. استخدم PNG أو JPG أو PDF.");
   420→   420→      return;
   421→   421→    }
   422→   422→    if (file.size > 10 * 1024 * 1024) {
   423→   423→      toast.error("حجم الملف يتجاوز 10 ميغابايت.");
   424→   424→      return;
   425→   425→    }
   426→   426→    const reader = new FileReader();
   427→   427→    reader.onload = () => {
   428→   428→      setImageUrl(reader.result as string);
   429→   429→      toast.success("تم تحميل الصورة — تُعرض على هذا الجهاز أثناء الجلسة الحالية.");
   430→   430→    };
   431→   431→    reader.readAsDataURL(file);
   432→   432→  }
   433→   433→
   434→   434→  return (
   435→   435→    <Card className="p-6">
   436→   436→      <div className="space-y-4">
   437→   437→        <div>
   438→   438→          <h3 className="font-bold text-sm mb-1">{t("schedule.uploadImage")}</h3>
   439→   439→          <p className="text-xs text-muted-foreground">{t("schedule.imageHint")}</p>
   440→   440→        </div>
   441→   441→        {imageUrl ? (
   442→   442→          <div className="space-y-3">
   443→   443→            <img
   444→   444→              src={imageUrl}
   445→   445→              alt="Schedule"
   446→   446→              className="w-full rounded-xl border border-border"
   447→   447→            />
   448→   448→            <Button variant="outline" size="sm" onClick={() => setImageUrl(null)}>
   449→   449→              تغيير الصورة
   450→   450→            </Button>
   451→   451→          </div>
   452→   452→        ) : (
   453→   453→          <label className="block">
   454→   454→            <div className="border-2 border-dashed border-border rounded-xl p-8 text-center cursor-pointer hover:border-primary/50 transition-colors">
