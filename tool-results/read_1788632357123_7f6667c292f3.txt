   113→function kindIcon(kind: string, className = "w-4 h-4") {
   114→  switch (kind) {
   115→    case "pdf": return <FileText className={className} />;
   116→    case "image": return <ImageIcon className={className} />;
   117→    case "video": return <Video className={className} />;
   118→    case "audio": return <Headphones className={className} />;
   119→    case "doc": case "ppt": return <File className={className} />;
   120→    case "text": return <MessageSquare className={className} />;
   121→    case "link": return <LinkIcon className={className} />;
   122→    default: return <File className={className} />;
   123→  }

   126→function formatDateAr(raw: string): string {
   127→  if (!raw) return "";
   128→  try {
   129→    const d = new Date(raw);
   130→    if (Number.isNaN(d.getTime())) return raw;
   131→    return d.toLocaleDateString("ar-DZ", { day: "numeric", month: "long", year: "numeric" });
   132→  } catch {
   133→    return raw;
   134→  }
   135→}

   357→        method: "POST",
   358→        headers: { "Content-Type": "application/json" },
   359→        body: JSON.stringify({
   360→          dayOfWeek,
   361→          startTime: e.time && e.time !== "—" ? e.time : "09:00",
   362→          endTime: "",
   363→          moduleName: course?.name ?? "",
